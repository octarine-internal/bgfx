cmake_minimum_required(VERSION 3.20)
project(bgfx-vulkan-tests LANGUAGES CXX)
find_package(Vulkan REQUIRED)
enable_testing()
add_executable(descriptor-allocator-vk descriptor_allocator_vk.cpp)
target_compile_features(descriptor-allocator-vk PRIVATE cxx_std_17)
target_link_libraries(descriptor-allocator-vk PRIVATE Vulkan::Vulkan)
add_test(NAME descriptor-allocator-unit COMMAND descriptor-allocator-vk)
add_test(NAME descriptor-allocator-vulkan COMMAND descriptor-allocator-vk --vulkan)
add_test(NAME descriptor-pool-legacy-limit COMMAND descriptor-allocator-vk --legacy)
